add_compile_options("$<$<COMPILE_LANGUAGE:C,CXX>:-Wall;-Wextra;-Wpedantic>")

option(APPLICATION_ENABLE_P2P "Enable application p2p workers" ON)

add_subdirectory(common)
if(DEFINED H755_CORE)
    if(H755_CORE STREQUAL "CM4")
        set(application_role io)
    else()
        set(application_role ui)
    endif()
    add_subdirectory(${application_role})
    target_sources(${PROJECT_NAME} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/${application_role}/main.cpp)
    target_link_libraries(${PROJECT_NAME} PRIVATE application::${application_role})
else()
    add_subdirectory(io)
    add_subdirectory(ui)
    target_sources(${PROJECT_NAME} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/main.cpp)
    target_link_libraries(${PROJECT_NAME} PRIVATE application::io application::ui)
endif()

if(BUILD_TESTING AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_subdirectory(tests)
endif()
